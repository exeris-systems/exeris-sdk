package eu.exeris.sdk.tck.corpus;

import eu.exeris.sdk.annotation.Action;
import eu.exeris.sdk.annotation.ExerisDomain;
import eu.exeris.sdk.annotation.Field;
import eu.exeris.sdk.annotation.NavMenu;
import eu.exeris.sdk.annotation.QueryParam;
import eu.exeris.sdk.annotation.Saga;
import eu.exeris.sdk.annotation.SagaStep;
import eu.exeris.sdk.annotation.Tab;
import eu.exeris.sdk.annotation.Validation;

/**
 * Corpus entity 1 of 2. Compiled as a unit with {@code Customer}, which points at it.
 *
 * <p>Every annotation attribute used here is one the SDK annotation actually declares, and none is
 * scheduled for removal. Neither half is a stylistic note. A corpus that compiles only because the
 * reader under test ignores what it does not recognise proves nothing, and the SDK has shipped
 * exactly that bug — a reader reading {@code @ExerisDomain(name = …)}, an attribute the annotation
 * has never declared. And a corpus is not merely read: a producer binding compiles it. Declaring
 * the tier through {@code tenantScoped} would have left this fixture failing to compile at 1.0.0,
 * when that boolean is removed — the kit breaking at exactly the freeze it exists to guard.
 *
 * <p>{@code version = 3} on the {@code @Saga} is the case behind the {@code SAGA} facet. Kernel
 * ADR-064 addresses a saga plan by name <em>and</em> version, and a side that skips this attribute
 * still emits well-formed metadata: the same source is version 3 to one side and version 1 — the
 * default an unread attribute leaves — to the other. It is not {@code 1} on purpose, because
 * {@code 1} is what a producer or reader that never looks at the attribute reports anyway.
 *
 * <p>The compensation attributes are the case behind the {@code SAGA_COMPENSATION} facet, and each
 * value is chosen so that a side which skips it, or drops it, says something else.
 * {@code compensationMaxRetries = 0} and the two {@code false}s are the values a boxed component
 * exists to carry: dropped, they read as "not declared", and the annotation defaults ({@code 5},
 * {@code true}) then apply. {@code compensationFailureHandler} is written as
 * {@code Order.CompensationEscalation}, a member type named through its declaring class, so a side
 * that carries the written form instead of the fully-qualified name javac reports disagrees with
 * every side that does not.
 *
 * <p>{@code @NavMenu}, the {@code @Tab}s and the {@code search} action are the cases behind the
 * {@code NAV_MENU}, {@code TABS} and {@code QUERY_PARAMS} facets. {@code order = 0} on the menu
 * entry and on the {@code summary} tab is a declared position that a side writing only
 * non-default values drops, and which then reads as the default {@code 999}. The {@code summary}
 * tab's attributes are declared on its first member only, and {@code totalCents} joins it by
 * name. {@code Order} declares no {@code @UI}, so a side that hangs tabs off {@code @UI} carries
 * none. On {@code search}, {@code term} is read under a query key that differs from its name, and
 * {@code limit} declares {@code required = false}, the opposite of the default; neither carries
 * {@code @ActionParam}, so a side that admits only {@code @ActionParam} parameters reads none.
 */
@ExerisDomain(
        module = "sales",
        path = "/orders",
        description = "A customer order.",
        dataScope = ExerisDomain.DataScope.TENANT
)
@NavMenu(
        group = "Sales",
        label = "Orders",
        order = 0,
        roles = {"SALES"},
        divider = NavMenu.Divider.AFTER
)
@Saga(
        name = "OrderFulfillmentSaga",
        version = 3,
        compensationTimeout = "PT2M",
        compensationMaxRetries = 0,
        compensationRetryDelay = "PT30S",
        continueCompensationOnFailure = false,
        compensationDlq = "sales.order-compensation.dlq",
        compensationFailureHandler = Order.CompensationEscalation.class,
        manualInterventionOnCompensationFailure = false
)
public class Order {

    @Field(label = "Reference", required = true)
    @Validation(minLength = 3, maxLength = 32, pattern = "^ORD-[0-9]+$")
    @Tab(name = "summary", label = "Summary", order = 0)
    private String reference;

    /**
     * {@code min = 0} is the case the 0.9.0 boxed-zero fix exists for: a zero bound is a real
     * non-negativity floor, and under a class-level {@code @JsonInclude(NON_DEFAULT)} Jackson 3
     * treats boxed zero as empty and drops it. A producer that loses this field's lower bound
     * emits a baseline that says "unbounded" where the source says "not negative".
     */
    @Field(label = "Total (cents)", required = true)
    @Validation(min = 0, max = 1000000)
    @Tab(name = "summary")
    private Long totalCents;

    @Field(label = "Note")
    @Tab(name = "notes", label = "Notes", lazy = true)
    private String note;

    @Action(name = "submit", label = "Submit", httpMethod = "POST")
    public void submit() {
    }

    @Action(name = "search", label = "Search", httpMethod = "GET")
    public void search(@QueryParam("q") String term,
                       @QueryParam(value = "limit", required = false) int limit) {
    }

    @SagaStep(order = 1, name = "charge", service = "payments", command = "charge")
    public void charge() {
    }

    /** The handler the saga names for a compensation that fails. */
    public static final class CompensationEscalation {
    }
}
