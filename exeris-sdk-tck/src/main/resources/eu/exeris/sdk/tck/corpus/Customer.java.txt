package eu.exeris.sdk.tck.corpus;

import eu.exeris.sdk.annotation.Blob;
import eu.exeris.sdk.annotation.ExerisDomain;
import eu.exeris.sdk.annotation.Field;
import eu.exeris.sdk.annotation.Relationship;
import eu.exeris.sdk.annotation.Relationship.RelationshipType;
import eu.exeris.sdk.annotation.UIGroup;
import eu.exeris.sdk.annotation.Validation;

/**
 * Corpus entity 2 of 3. Compiled as a unit with {@code Order} and {@code Product}.
 *
 * <p>The {@code @Relationship} here is not decoration. Its cardinality attribute is named
 * {@code relationshipType}, and a producer that reads it under a different key silently hands back
 * the builder default — which is how a {@code ONE_TO_MANY} side came to emit the foreign key,
 * index, constraint and finder that belong on the other side. The defect was invisible because
 * both sides still produced a well-formed baseline.
 *
 * <p>{@code displayField = "reference"} is not {@code name}, the value a side that never reads
 * the attribute reports. {@code orders} declares {@code cascadeDelete} alone, so a side that skips
 * the two booleans reads no cascade, and one that collapses them into {@code ALL} reads the wrong
 * one. {@code lastOrderId} is the explicit-key style: a {@code UUID} field whose
 * {@code targetEntity} names {@code Order}, so a side that takes the target from the field type
 * reads {@code UUID}.
 *
 * <p>{@code permissions} without a {@code @RouteAccess} is the entity-level case of a route that
 * declares its scopes alone: the metadata carries the list and no {@code routeAccess}.
 *
 * <p>The {@code @UIGroup}s are the case behind the {@code UI_GROUPS} facet. The {@code identity}
 * group's label and order are declared on its first member; {@code displayName} restates the label,
 * which does not change it, and is the first to declare {@code collapsed}. {@code media} declares
 * {@code collapsible = false}, the opposite of the annotation default, and is declared second with
 * the lower order, so a side that sorts the groups reorders them. {@code Customer} declares no
 * {@code @UI}, so a side that hangs groups off {@code @UI} carries none.
 *
 * <p>The {@code @Blob}s are the case behind the {@code BLOBS} facet: {@code avatar} beside
 * {@code @Field} with two media types, {@code contract} on a field that carries no {@code @Field},
 * so a side that reads {@code @Blob} only beside {@code @Field} drops it. The entity is
 * {@code TENANT}-scoped, so neither is the {@code GLOBAL} combination the platform refuses.
 */
@ExerisDomain(
        module = "sales",
        path = "/customers",
        description = "A person or organisation that places orders.",
        dataScope = ExerisDomain.DataScope.TENANT,
        permissions = {"customer:read"}
)
public class Customer {

    @Field(label = "E-mail", required = true, unique = true)
    @Validation(email = true, maxLength = 254)
    @UIGroup(name = "identity", label = "Identity", order = 2)
    private String email;

    @Field(label = "Display name", required = true)
    @Validation(minLength = 1, maxLength = 120)
    @UIGroup(name = "identity", label = "Restated", collapsed = true)
    private String displayName;

    @Field(label = "Avatar")
    @Blob(contentTypes = {"image/png", "image/jpeg"})
    @UIGroup(name = "media", label = "Media", order = 0, collapsible = false)
    private String avatar;

    @Blob(container = "contracts")
    private String contract;

    @Relationship(
            targetEntity = Order.class,
            displayField = "reference",
            relationshipType = RelationshipType.ONE_TO_MANY,
            mappedBy = "customer",
            cascadeDelete = true)
    private java.util.List<Order> orders;

    @Relationship(targetEntity = Order.class, displayField = "reference")
    private java.util.UUID lastOrderId;
}
